/* =========================================================

   COUNTDOWN

   ========================================================= */


function updateCountdown() {

    const target = new Date(DATA.electionDate).getTime();

    const now = Date.now();

    const difference = target - now;


    if (difference <= 0) {

        byId("countdown").textContent = "Election day";

        return;

    }


    const days = Math.floor(difference / 86400000);

    const hours = Math.floor(

        (difference % 86400000) / 3600000

    );

    const minutes = Math.floor(

        (difference % 3600000) / 60000

    );


    byId("countdown").textContent =

        days + "d " +

        hours + "h " +

        minutes + "m";

}

   PARLIAMENT SVG

   ========================================================= */


function renderParliament() {

    const svg = byId("parliamentSVG");

    const seats = [];


    /*

       123 seats arranged in stable horizontal rows.

       This deliberately avoids complex SVG arcs so the layout

       remains predictable on desktop and mobile browsers.

    */


    let seatNumber = 0;


    for (let row = 0; row < 8; row++) {

        const count =

            row < 7 ? 16 : 11;


        const spacing = 50;

        const width =

            (count - 1) * spacing;


        const startX =

            (1000 - width) / 2;


        const y =

            75 + row * 58;


        for (

            let column = 0;

            column < count;

            column++

        ) {

            if (

                seatNumber >=

                DATA.totalSeats

            ) {

                break;

            }


            seats.push({

                x:

                    startX +

                    column * spacing,

                y: y

            });


            seatNumber++;

        }

    }


    const allocation = [];


    DATA.parties.forEach(function(party) {

        for (

            let i = 0;

            i < party.seats;

            i++

        ) {

            allocation.push(party);

        }

    });


    let svgContent = "";


    seats.forEach(function(seat, index) {

        const party =

            allocation[index];


        svgContent +=

            "<circle " +

                "cx='" +

                    seat.x +

                "' " +

                "cy='" +

                    seat.y +

                "' " +

                "r='13' " +

                "fill='" +

                    party.color +

                "' " +

                "stroke='#ffffff' " +

                "stroke-width='1'>" +


                "<title>" +

                    party.name +

                    " — projected seat " +

                    (index + 1) +

                "</title>" +


            "</circle>";

    });


    svgContent +=

        "<text " +

            "x='500' " +

            "y='555' " +

            "text-anchor='middle' " +

            "fill='#657581' " +

            "font-size='12'>" +

            "123 projected seats · majority = 62" +

        "</text>";


    svg.innerHTML =

        svgContent;


    const totalSeats =

        DATA.parties.reduce(

            function(total, party) {

                return total +

                    party.seats;

            },

            0

        );


    byId("majorityFill").style.width =

        (

            (totalSeats /

                DATA.totalSeats) *

            100

        ).toFixed(2) +

        "%";


    byId("seatLegend").innerHTML =

        DATA.parties

        .slice()

        .sort(function(a, b) {

            return b.seats -

                a.seats;

        })

        .map(function(party) {

            return (

                "<div class='legend-row'>" +


                    "<span " +

                        "class='legend-square' " +

                        "style='background:" +

                            party.color +

                        ";'>" +

                    "</span>" +


                    "<span>" +

                        party.name +

                    "</span>" +


                    "<span class='legend-seats'>" +

                        party.seats +

                    "</span>" +


                "</div>"

            );

        }).join("");

}

function updateBuilder() {

    const selected =

        all("#checkGrid input:checked")

        .map(function(input) {

            return input.value;

        });


    const seats =

        selected.reduce(

            function(total, id) {


                const party =

                    DATA.parties.find(

                        function(p) {

                            return p.id === id;

                        }

                    );


                return total +

                    (

                        party

                            ? party.seats

                            : 0

                    );

            },

            0

        );


    const number =

        byId("resultNumber");


    number.textContent =

        seats;


    number.classList.toggle(

        "good",

        seats >= DATA.majority

    );


    if (selected.length === 0) {


        byId("resultMessage")

            .textContent =

            "Select parties to test a government.";


    } else if (

        seats >= DATA.majority

    ) {


        const difference =

            seats - DATA.majority;


        byId("resultMessage")

            .textContent =

            difference +

            " seat" +

            (

                difference === 1

                    ? ""

                    : "s"

            ) +

            " above the majority line.";


    } else {


        const difference =

            DATA.majority - seats;


        byId("resultMessage")

            .textContent =

            difference +

            " seat" +

            (

                difference === 1

                    ? ""

                    : "s"

            ) +

            " short of a majority.";

    }


    byId("resultMeter").style.width =

        Math.min(

            100,

            (

                seats /

                DATA.majority

            ) * 100

        ).toFixed(2) +

        "%";

}
