/*

   This is intentionally a simple proportional model.


   It is NOT a reproduction of Spain's full provincial

   D'Hondt allocation.


   The purpose here is to make the live dashboard respond

   consistently to polling movement.

*/


function projectSeats() {


    const parties =

        electionData.parties;


    const eligible =

        parties.filter(

            party => party.percentage >= 3

        );


    const totalPolling =

        eligible.reduce(

            (sum, party) =>

                sum + party.percentage,

            0

        );



    if (totalPolling <= 0) {


        parties.forEach(

            party => party.seats = 0

        );


        return;


    }



    const allocations =

        eligible.map(party => {


            const raw =

                party.percentage /

                totalPolling *

                electionData.totalSeats;


            return {


                party,


                raw,


                seats: Math.floor(raw),


                remainder:

                    raw - Math.floor(raw)


            };


        });



    let allocated =

        allocations.reduce(

            (sum, item) =>

                sum + item.seats,

            0

        );



    allocations.sort(

        (a, b) =>

            b.remainder -

            a.remainder

    );



    let index = 0;



    while (

        allocated <

        electionData.totalSeats

    ) {


        allocations[index].seats++;


        allocated++;


        index++;


        if (

            index >=

            allocations.length

        ) {

            index = 0;

        }


    }



    parties.forEach(

        party => party.seats = 0

    );



    allocations.forEach(

        item => {

            item.party.seats =

                item.seats;

        }

    );


}



/* ============================================================

   COALITION CALCULATION

============================================================ */


function calculateCoalitions() {


    return electionData

        .coalitionDefinitions

        .map(definition => {


            const seats =

                definition.parties.reduce(

                    (sum, partyId) => {


                        const party =

                            electionData.parties.find(

                                p => p.id === partyId

                            );


                        return sum +

                            (party

                                ? party.seats

                                : 0);


                    },

                    0

                );



            return {


                ...definition,


                seats,


                viable:

                    seats >=

                    electionData.majorityNeeded


            };


        });


}

/* ============================================================

   LIVE POLLING MODEL

============================================================ */


function updateLiveModel() {


    if (state.paused) {

        return;

    }



    state.tick++;



    electionData.parties.forEach(

        (party, index) => {


            const baseline =

                party.baseline;



            /*

               Very small movement.


               - random component

               - tiny cyclical component

               - mean reversion toward baseline


               This prevents the model from drifting

               indefinitely.

            */


            const randomMovement =

                (Math.random() - 0.5) *

                0.16;



            const cycle =

                Math.sin(

                    state.tick / 7 +

                    index * 1.4

                ) *

                0.035;



            const meanReversion =

                (baseline -

                    party.percentage) *

                0.07;



            let next =

                party.percentage +

                randomMovement +

                cycle +

                meanReversion;



            /*

               Prevent extremely small parties

               from suddenly becoming huge.

            */


            next =

                clamp(

                    next,

                    Math.max(

                        0.5,

                        baseline - 2.5

                    ),

                    baseline + 2.5

                );



            party.percentage =

                Number(

                    next.toFixed(1)

                );


        }

    );



    state.updated =

        new Date();



    projectSeats();



    const ranking =

        [...electionData.parties]

            .sort(

                (a, b) =>

                    b.percentage -

                    a.percentage

            );



    const leader =

        ranking[0];



    state.history.push({


        time:

            state.updated,


        value:

            leader.percentage,


        leader:

            leader.shortName


    });



    /*

       Keep the chart reasonably small.

    */


    if (

        state.history.length >

        40

    ) {


        state.history.shift();


    }



    /*

       Don't create a feed entry every

       single second. Every third update

       is enough.

    */


    if (

        state.tick === 1 ||

        state.tick % 3 === 0

    ) {


        addEvent(

            `${leader.shortName} `

            + `at ${leader.percentage.toFixed(1)}%`

        );


    }



    render();


}

/* ============================================================

   COUNTDOWN

============================================================ */


function updateCountdown() {


    const target =

        new Date(

            "2026-11-29T00:00:00"

        );



    const now =

        new Date();



    const difference =

        target - now;



    if (

        difference <= 0

    ) {


        $("#countdown")

            .textContent =

            "Election day";


        return;


    }



    const days =

        Math.floor(

            difference /

            86400000

        );



    const hours =

        Math.floor(

            (

                difference %

                86400000

            ) /

            3600000

        );



    const minutes =

        Math.floor(

            (

                difference %

                3600000

            ) /

            60000

        );



    $("#countdown")

        .textContent =

        `${days}d ${hours}h ${minutes}m`;


}

function init() {


    /*

       Calculate the first seat projection

       before anything is displayed.

    */


    projectSeats();



    /*

       Initial history point.

    */


    const initialLeader =

        [...electionData.parties]

            .sort(

                (a, b) =>

                    b.percentage -

                    a.percentage

            )[0];



    state.history.push({


        time: new Date(),


        value:

            initialLeader.percentage,


        leader:

            initialLeader.shortName


    });



    addEvent(

        "Live model started from supplied baseline"

    );



    render();


    updateCountdown();



    /*

       Countdown only needs to update once

       per minute.

    */


    setInterval(

        updateCountdown,

        60000

    );



    /*

       Main live polling movement.


       Five seconds is fast enough to feel

       alive, while the movement itself stays

       deliberately subtle.

    */


    setInterval(

        updateLiveModel,

        5000

    );


}
